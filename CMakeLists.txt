cmake_minimum_required(VERSION 3.24)
if(NOT DEFINED CMAKE_CUDA_ARCHITECTURES)
    set(CMAKE_CUDA_ARCHITECTURES 86 120 CACHE STRING "GPU architectures")
endif()
project(qwen_tuning LANGUAGES CXX CUDA)
find_package(CUDAToolkit REQUIRED)
execute_process(COMMAND git rev-parse HEAD WORKING_DIRECTORY "${CMAKE_SOURCE_DIR}" OUTPUT_VARIABLE REVISION_HEAD OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
execute_process(COMMAND git rev-parse --path-format=absolute --git-path index WORKING_DIRECTORY "${CMAKE_SOURCE_DIR}" OUTPUT_VARIABLE REAL_INDEX OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
set(REVISION_INDEX "${CMAKE_BINARY_DIR}/revision.index")
file(COPY_FILE "${REAL_INDEX}" "${REVISION_INDEX}")
execute_process(COMMAND "${CMAKE_COMMAND}" -E env "GIT_INDEX_FILE=${REVISION_INDEX}" git add -A WORKING_DIRECTORY "${CMAKE_SOURCE_DIR}" COMMAND_ERROR_IS_FATAL ANY)
execute_process(COMMAND "${CMAKE_COMMAND}" -E env "GIT_INDEX_FILE=${REVISION_INDEX}" git diff --cached --no-color --no-ext-diff HEAD WORKING_DIRECTORY "${CMAKE_SOURCE_DIR}" OUTPUT_VARIABLE REVISION_DIFF COMMAND_ERROR_IS_FATAL ANY)
file(REMOVE "${REVISION_INDEX}")
set(REVISION_TEXT "${REVISION_HEAD}")
if(REVISION_DIFF)
    string(APPEND REVISION_TEXT "\n\n${REVISION_DIFF}")
endif()
file(WRITE "${CMAKE_BINARY_DIR}/revision.txt" "${REVISION_TEXT}")
file(READ "${CMAKE_BINARY_DIR}/revision.txt" REVISION_HEX HEX)
string(REGEX REPLACE "(..)" "0x\\1," REVISION_BYTES "${REVISION_HEX}")
file(WRITE "${CMAKE_BINARY_DIR}/revision.cpp" "#include <cstddef>\nextern const unsigned char qwen_revision[] = {${REVISION_BYTES}};\nextern const size_t qwen_revision_size = sizeof(qwen_revision);\n")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS native/experiment.h native/qwen.cu)
add_executable(qwen-tune native/qwen.cu native/tensorboard.cpp "${CMAKE_BINARY_DIR}/revision.cpp")
set_target_properties(qwen-tune PROPERTIES CXX_STANDARD 20 CUDA_STANDARD 20)
target_link_libraries(qwen-tune PRIVATE CUDA::cublas CUDA::cuda_driver)
target_compile_options(qwen-tune PRIVATE "$<$<COMPILE_LANGUAGE:CUDA>:-O3;-lineinfo;--default-stream=per-thread>")
if(MSVC)
    set_property(TARGET qwen-tune PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
